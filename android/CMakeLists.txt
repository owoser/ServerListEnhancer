cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_LIST_DIR}/../version.json" SLE_VERSION_JSON)
string(JSON SLE_VERSION GET "${SLE_VERSION_JSON}" version)
string(JSON SLE_LL_VERSION GET "${SLE_VERSION_JSON}" leviLaminaVersion)
project(server_list_enhancer_android VERSION "${SLE_VERSION}" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "Keep pinned dependency checkouts unchanged")
# Match the exact Preloader submodule used by LeviLauncher Android v1.6.0.
FetchContent_Declare(preloader_android
    GIT_REPOSITORY https://github.com/LiteLDev/preloader-android.git
    GIT_TAG 5d108da350ce3e7f13f51415f81e65344ecd5d14
    SOURCE_SUBDIR cmake/preloader-sdk)
FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
FetchContent_Declare(boost_pfr GIT_REPOSITORY https://github.com/boostorg/pfr.git GIT_TAG boost-1.88.0)
FetchContent_Declare(magic_enum GIT_REPOSITORY https://github.com/Neargye/magic_enum.git GIT_TAG v0.9.7)
FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.2.0)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(preloader_android nlohmann_json boost_pfr magic_enum fmt)
if(ANDROID)
    # NDK r30 no longer exposes global malloc/free through fmt 11.2 transitive includes.
    target_compile_options(fmt PRIVATE -include stdlib.h)
endif()
set(SLE_DEPS nlohmann_json::nlohmann_json Boost::pfr magic_enum::magic_enum fmt::fmt)
if(NOT ANDROID)
    add_executable(sle_android_config src/GenerateConfig.cpp)
    target_include_directories(sle_android_config PRIVATE src "${preloader_android_SOURCE_DIR}/include")
    target_compile_definitions(sle_android_config PRIVATE PL_CONFIG_NO_RUNTIME)
    target_link_libraries(sle_android_config PRIVATE ${SLE_DEPS})
    add_custom_target(sle_generate_config
        COMMAND sle_android_config "${CMAKE_BINARY_DIR}/generated-config"
        DEPENDS sle_android_config VERBATIM)
    enable_testing()
    add_executable(sle_android_model_test ../tests/android_model_test.cpp)
    target_include_directories(sle_android_model_test PRIVATE src ../src-client ../src)
    target_link_libraries(sle_android_model_test PRIVATE nlohmann_json::nlohmann_json)
    add_test(NAME android_model COMMAND sle_android_model_test)
    add_executable(sle_android_touch_test ../tests/android_touch_test.cpp)
    target_include_directories(sle_android_touch_test PRIVATE src)
    add_test(NAME android_touch COMMAND sle_android_touch_test)
    add_executable(sle_android_ime_test ../tests/android_ime_test.cpp)
    target_include_directories(sle_android_ime_test PRIVATE src ../src-client)
    add_test(NAME android_ime COMMAND sle_android_ime_test)
    return()
endif()
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This port supports arm64-v8a only")
endif()
set(LEVI_PACKAGE_CONFIG_DIR "" CACHE PATH "Host-generated config directory")
foreach(file config.json config.schema.json)
    if(NOT EXISTS "${LEVI_PACKAGE_CONFIG_DIR}/${file}")
        message(FATAL_ERROR "Run scripts/package-android.ps1 to generate ${file}")
    endif()
endforeach()
add_subdirectory("${preloader_android_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/_preloader_android")
add_library(server_list_enhancer SHARED src/AndroidMod.cpp src/AndroidPlatform.cpp src/HomeUi.cpp
    ../src-client/ServerStore.cpp ../src/common/BuiltinImages.cpp)
target_include_directories(server_list_enhancer PRIVATE src ../src ../src-client
    "${preloader_android_SOURCE_DIR}/include")
target_compile_definitions(server_list_enhancer PRIVATE SLE_CLIENT)
target_compile_options(server_list_enhancer PRIVATE -Wall -Wextra -fvisibility=hidden -fvisibility-inlines-hidden)
# Link for DT_NEEDED and strict symbol checks; the launcher supplies this runtime at installation.
target_link_options(server_list_enhancer PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(server_list_enhancer PRIVATE preloader ${SLE_DEPS} log android)
set(MOD_ID server_list_enhancer)
set(MOD_NAME "ServerListEnhancer")
set(MOD_AUTHOR "OWOSER Dev")
set(MOD_VERSION "${PROJECT_VERSION}")
configure_file(manifest.json.in manifest.json @ONLY)
